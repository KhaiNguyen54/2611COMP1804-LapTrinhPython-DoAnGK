\section{Các nhận xét chính}\label{sec:nhanxet}
Từ bảy phép phân tích ở Chương~\ref{ch:phantich} và kết quả mô hình ở Chương~\ref{ch:mohinh}, báo cáo rút ra chín nhận xét; số liệu và nguồn được dẫn kèm.
\begin{enumerate}
  \item \textbf{Loại nhà phân biệt giá mạnh.} Trung vị của nhà riêng là 1,080,000 AUD, gấp 1.93 lần căn hộ (560,000); nhà liền kề 846,750 (Phép~2; \texttt{price\_by\_type.csv}, \bs{P03}).
  \item \textbf{Vị trí tạo chênh lệch lớn.} Southern Metropolitan (1,250,000) cao hơn Western Metropolitan (793,000) khoảng 1.6 lần; Albert Park, suburb đắt nhất trong nhóm từ 30 giao dịch (1,900,000), gấp 2.1 lần trung vị chung (903,000) (Phép~3, 5; \bs{P04, P06}).
  \item \textbf{Giá tăng theo số phòng.} Nhà 4 phòng có trung vị 1,285,000, gấp 3.3 lần nhà 1 phòng (385,000); Pearson giữa giá và số phòng là 0.497 (Phép~4, 6; \bs{P05, P07}).
  \item \textbf{Khoảng cách ảnh hưởng nhưng số liệu gộp che mờ.} Pearson chung $-$0.163; nhà riêng $-$0.329, căn hộ $-$0.051 (Phép~6; \bs{P07}). Việc căn hộ rẻ và gần trung tâm hơn là một cách giải thích, chưa được kiểm định.
  \item \textbf{Giá lệch phải.} Độ lệch 2.24, còn 0.18 sau khi lấy log; trung bình (1,075,684) cao hơn trung vị (903,000) 19\% (Phép~1; \bs{L04, P01}). Vì vậy báo cáo dùng trung vị để so sánh và log giá để huấn luyện Random Forest.
  \item \textbf{Ngoại lai diện tích làm sai lệch tương quan.} Pearson của \texttt{BuildingArea} với giá là 0.092 khi còn 8 dòng trên 1,000 m\textsuperscript{2}, và 0.556 sau khi loại chúng; \texttt{Landsize} tăng từ 0.038 lên 0.132 sau khi loại 26 dòng trên 10,000 m\textsuperscript{2} (Phép~6; \bs{L01, P07}).
  \item \textbf{Dữ liệu thiếu và giá trị 0 mang thông tin.} Sau làm sạch, \texttt{BuildingArea} thiếu 47.7\% và \texttt{YearBuilt} thiếu 39.6\% (\bs{L03}). Có 1,939 dòng \texttt{Landsize} bằng 0, trong đó 84.0\% là căn hộ; các dòng này có giá trung vị 560,000 so với 975,000 ở dòng có đất (\bs{L01, L02, L05}). Giá trị 0 vì vậy gắn với loại nhà, không phải nhiễu ngẫu nhiên (Chương~\ref{ch:lamsach}).
  \item \textbf{Trung vị theo tháng chịu ảnh hưởng của cơ cấu giao dịch.} Tháng 7 thấp nhất (773,500), nhưng tháng 7/2017 chỉ có 19.3\% giao dịch ở Southern Metropolitan, so với 29.5\% và 30.0\% ở hai tháng kề bên (Phép~7; \bs{P08}). Không kết luận được giá giảm.
  \item \textbf{Mô hình vượt xa baseline nhưng yếu ở phân khúc cao.} MAE test của Random Forest là 156,203 AUD, thấp hơn baseline (439,350) 64.4\%, $R^2$ 0.8225. Ở phân khúc từ 1,500,000 AUD, MAE là 368,100 và dự đoán thấp hơn thực tế trung bình 231,515 AUD (Chương~\ref{ch:mohinh}; \texttt{model\_metrics.csv}, \bm{M01, M03}).
\end{enumerate}
